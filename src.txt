<script setup lang="ts">
import { ref, onMounted, onUnmounted, computed } from 'vue';
import { gameStore } from './store/gameStore';
import { HIRAGANA_DATA } from './lib/constants';
import { shuffleArray, isSameOrder } from './lib/utils';
import type { GameState, HiraganaChar } from './types';
import AppToolbar from './components/AppToolbar.vue';
import HiraganaItem from './components/HiraganaItem.vue';
import AppModal from './components/AppModal.vue';

// 1. Hubungkan State Zustand ke Reactivity Vue
const state = ref<GameState>(gameStore.getState());

const unsubscribe = gameStore.subscribe((newState) => {
  state.value = newState;
});

let timerInterval: ReturnType<typeof setInterval> | null = null;
const isModalOpen = ref(false);

// 2. Lifecycle Hooks
onMounted(() => {
  state.value = gameStore.getState();
});

onUnmounted(() => {
  unsubscribe();
  if (timerInterval) clearInterval(timerInterval);
});

// 3. Helper Functions
const isSelected = (id: string) => state.value.selectedChars.some(c => c.id === id);
const isDisabled = (id: string) => !isSelected(id) && state.value.selectedChars.length >= state.value.maxCards;

const handleToggleChar = (char: HiraganaChar) => {
  gameStore.getState().toggleChar(char);
};

// 4. Main Actions
const handleStart = () => {
  // Cegah mulai jika belum ada huruf yang dipilih
  if (state.value.selectedChars.length === 0) {
    isModalOpen.value = true;
    return;
  }
  
  let initialCards: HiraganaChar[] = [];

  if (state.value.isShuffleEnabled) {
    // JIKA MODE ACAK ON: Acak huruf
    initialCards = shuffleArray(state.value.selectedChars);
    
    // Cek Anti-Duplikat: Jika kartu sebelumnya ada, dan hasil acak baru SAMA,
    // maka acak ulang terus sampai berbeda.
    if (state.value.activeCards.length > 0) {
      while (isSameOrder(initialCards, state.value.activeCards)) {
        initialCards = shuffleArray(state.value.selectedChars);
      }
    }
  } else {
    // JIKA MODE ACAK OFF: Tampilkan urut sesuai checklist
    initialCards = [...state.value.selectedChars];
  }
  
  gameStore.getState().setActiveCards(initialCards);
  gameStore.getState().startGame();

  // Jalankan Timer
  if (timerInterval) clearInterval(timerInterval);
  timerInterval = setInterval(() => {
    if (!gameStore.getState().isPlaying) {
      if (timerInterval) clearInterval(timerInterval);
      return;
    }
    gameStore.getState().tickTimer();
  }, 1000);
};

const handleReset = () => {
  if (timerInterval) clearInterval(timerInterval);
  gameStore.getState().resetGame();
};

// 5. Computed Styles
const fontClass = computed(() => {
  const count = state.value.activeCards.length;
  if (count <= 10) return 'text-5xl md:text-6xl';
  if (count <= 25) return 'text-4xl md:text-5xl';
  return 'text-3xl md:text-4xl';
});

const gridClass = computed(() => {
  if (state.value.useCardStyle) {
    return 'grid-cols-4 sm:grid-cols-5 md:grid-cols-6 lg:grid-cols-8';
  } else {
    return 'grid-cols-5 sm:grid-cols-6 md:grid-cols-8 lg:grid-cols-10';
  }
});
</script>

<template>

    <!-- Header -->
    <header class="text-center mb-6">
      <h1 class="text-2xl md:text-3xl font-bold text-slate-800 mb-1">Hiragana Drill</h1>
      <p class="text-slate-500 text-sm">Latihan membaca cepat</p>
      
      <!-- Timer Display -->
      <div 
        v-if="state.isPlaying" 
        class="mt-3 text-3xl font-mono font-bold inline-block px-4 py-1 rounded-full"

      >
        ⏱ {{ state.timeLeft }} dtk
      </div>
    </header>

    <!-- Toolbar Kontrol -->
    <AppToolbar 
      :state="state"
      :on-start="handleStart"
      @update:max-cards="(val) => gameStore.getState().setMaxCards(val)"

      @toggle-card-style="() => gameStore.getState().toggleCardStyle()"
      @toggle-shuffle="() => gameStore.getState().toggleShuffle()"
    />

            ]"
          >
            {{ char.char }}

          :use-card-style="state.useCardStyle"
          :class="fontClass"
        />
      </div>

      <!-- AREA 3: Waktu Habis (Tampil saat FINISHED) -->
      <div v-if="state.isPlaying && state.timeLeft === 0" class="text-center mt-12 animate-in fade-in zoom-in duration-300">
        <div class="inline-block bg-white p-8 rounded-2xl shadow-lg">
          <h2 class="text-3xl font-bold text-slate-800 mb-2">Waktu Habis! ⏰</h2>

          >

  </div>

// src/lib/utils.ts
import type { HiraganaChar } from '../types';

export const shuffleArray = (array: HiraganaChar[]): HiraganaChar[] => {
  const newArr = [...array];
  for (let i = newArr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [newArr[i], newArr[j]] = [newArr[j], newArr[i]];
  }
  return newArr;
};

export const isSameOrder = (arr1: HiraganaChar[], arr2: HiraganaChar[]): boolean => {
  if (arr1.length !== arr2.length) return false;
  return arr1.every((val, index) => val.id === arr2[index].id);
};

// src/store/gameStore.ts
import { createStore } from 'zustand/vanilla';
import type { GameState } from '../types';
import { HIRAGANA_DATA } from '../lib/constants';

export const gameStore = createStore<GameState>((set, get) => ({
  maxCards: 5,
  selectedChars: HIRAGANA_DATA.slice(0, 5),
  activeCards: [],
  showRomaji: true,
  useCardStyle: true,
  isAutoShuffle: false,
  timeInput: 60,
  timeLeft: 60,
  isPlaying: false,

  setMaxCards: (num) => set((state) => {
    const newSelected = state.selectedChars.slice(0, num);
    return { maxCards: num, selectedChars: newSelected };
  }),

  toggleChar: (char) => set((state) => {
    const isSelected = state.selectedChars.find((c) => c.id === char.id);
    if (isSelected) {
      return { selectedChars: state.selectedChars.filter((c) => c.id !== char.id) };
    }
    if (state.selectedChars.length >= state.maxCards) return state;
    return { selectedChars: [...state.selectedChars, char] };
  }),

  toggleRomaji: () => set((state) => ({ showRomaji: !state.showRomaji })),
  toggleCardStyle: () => set((state) => ({ useCardStyle: !state.useCardStyle })),
  toggleAutoShuffle: () => set((state) => ({ isAutoShuffle: !state.isAutoShuffle })),

  setTimeInput: (time) => set({ timeInput: time, timeLeft: time }),
  setActiveCards: (cards) => set({ activeCards: cards }),

  startGame: () => {
    const state = get();
    if (state.selectedChars.length === 0) return false;
    set({ isPlaying: true, timeLeft: state.timeInput });
    return true;
  },

  tickTimer: () => set((state) => {
    if (state.timeLeft <= 1) {
      return { timeLeft: 0, isPlaying: false };
    }
    return { timeLeft: state.timeLeft - 1 };
  }),

  resetGame: () => set((state) => ({ 
    isPlaying: false, 
    timeLeft: state.timeInput,
    activeCards: [] 
  })),
}));
