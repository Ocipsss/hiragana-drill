<script setup lang="ts">
import { ref, onMounted, onUnmounted, computed } from 'vue';
import { gameStore } from './store/gameStore';
import { HIRAGANA_DATA } from './lib/constants';
import { shuffleArray, isSameOrder, getFilteredWords, formatTime } from './lib/utils';
import { saveHistory, getHistory, checkPersonalBest, clearAllHistory, deleteHistoryEntry } from './lib/db';
import { toCSV, toJSON, downloadFile, generateFilename } from './lib/export';
import type { GameState, StudyItem, RowType } from './types';
import { RotateCcw, Filter } from 'lucide-vue-next';
import AppToolbar from './components/AppToolbar.vue';
import HiraganaItem from './components/HiraganaItem.vue';
import AppModal from './components/AppModal.vue';
import SelectModal from './components/SelectModal.vue';
import FinishScreen from './components/FinishScreen.vue';
import HistoryModal from './components/HistoryModal.vue';
import ExportModal from './components/ExportModal.vue';
import ConfirmModal from './components/ConfirmModal.vue';
import RowFilterModal from './components/RowFilterModal.vue';

const state = ref<GameState>(gameStore.getState());

const unsubscribe = gameStore.subscribe((newState) => {
  state.value = newState;
});

let timerInterval: ReturnType<typeof setInterval> | null = null;
const isModalOpen = ref(false);
const isFilterModalOpen = ref(false);
const isHistoryModalOpen = ref(false);
const isExportModalOpen = ref(false);
const isRestartModalOpen = ref(false);
const isRowFilterModalOpen = ref(false);

onMounted(async () => {
  state.value = gameStore.getState();
  await gameStore.getState().loadHistory();
});

onUnmounted(() => {
  unsubscribe();
  if (timerInterval) clearInterval(timerInterval);
});

const currentData = computed(() => {
  return state.value.mode === 'char' ? HIRAGANA_DATA : getFilteredWords(state.value.wordLengthFilter);
});

const maxAvailable = computed(() => currentData.value.length);

const isSelected = (id: string) => state.value.selectedItems.some(c => c.id === id);
const isDisabled = (id: string) => {
  const limit = state.value.mode === 'char' ? state.value.maxChars : state.value.maxWords;
  return !isSelected(id) && state.value.selectedItems.length >= limit;
};

const handleToggleItem = (item: StudyItem) => {
  gameStore.getState().toggleItem(item);
};

// Handler untuk multi-select filter
const handleToggleRow = (row: RowType) => {
  gameStore.getState().toggleRowFilter(row);
};

const startTimer = () => {
  if (timerInterval) clearInterval(timerInterval);
  timerInterval = setInterval(() => {
    if (!gameStore.getState().isPlaying) {
      if (timerInterval) clearInterval(timerInterval);
      return;
    }
    gameStore.getState().tickTimer();
  }, 16);
};

const handleStart = () => {
  if (state.value.selectedItems.length === 0) {
    isModalOpen.value = true;
    return;
  }
  
  let initialItems: StudyItem[] = [];
  if (state.value.isShuffleEnabled) {
    initialItems = shuffleArray(state.value.selectedItems);
    if (state.value.activeItems.length > 0) {
      while (isSameOrder(initialItems, state.value.activeItems)) {
        initialItems = shuffleArray(state.value.selectedItems);
      }
    }
  } else {
    initialItems = [...state.value.selectedItems];
  }
  
  gameStore.getState().setActiveItems(initialItems);
  gameStore.getState().startGame();
  startTimer();
};

const handleRestartClick = () => {
  isRestartModalOpen.value = true;
};

const confirmRestart = () => {
  isRestartModalOpen.value = false;
  
  if (state.value.isShuffleEnabled) {
    let newItems = shuffleArray(state.value.selectedItems);
    while (isSameOrder(newItems, state.value.activeItems)) {
      newItems = shuffleArray(state.value.selectedItems);
    }
    gameStore.getState().setActiveItems(newItems);
  }
  
  gameStore.getState().restartGame();
  startTimer();
};

const handleFinish = async () => {
  if (timerInterval) clearInterval(timerInterval);
  const currentState = gameStore.getState();
  if (!currentState.isPlaying) return;

  const now = Date.now();
  const realDurationMs = now - currentState.startTimestamp - currentState.pausedAccumMs;
  const itemCount = currentState.activeItems.length;
  const avgMsPerItem = itemCount > 0 ? realDurationMs / itemCount : 0;
  const wordFilter = currentState.mode === 'word' ? currentState.wordLengthFilter : undefined;

  const isPB = await checkPersonalBest(
    currentState.mode, 
    currentState.timerMode, 
    itemCount, 
    wordFilter, 
    avgMsPerItem
  );

  const entry = {
    timestamp: Date.now(),
    mode: currentState.mode,
    timerMode: currentState.timerMode,
    itemCount,
    wordFilter,
    durationMs: realDurationMs,
    avgMsPerItem,
    items: JSON.parse(JSON.stringify(currentState.activeItems)), 
    isPersonalBest: isPB
  };

  const saved = await saveHistory(entry);
  const updatedHistory = await getHistory();

  gameStore.getState().stopAndSetRecord(saved);
  gameStore.getState().setHistoryData(updatedHistory, saved);
};

const handleTogglePause = () => {
  gameStore.getState().togglePause();
};

const handleReset = () => {
  if (timerInterval) clearInterval(timerInterval);
  gameStore.getState().resetGame();
};

const handleDeleteHistory = async (id: number) => {
  await deleteHistoryEntry(id);
  const updatedHistory = await getHistory();
  gameStore.getState().setHistoryData(updatedHistory, state.value.lastRecord);
};

const handleClearAllHistory = async () => {
  await clearAllHistory();
  gameStore.getState().setHistoryData([], null);
};

const handleExport = (payload: { format: 'csv' | 'json'; filteredHistory: any[] }) => {
  const { format, filteredHistory } = payload;
  if (format === 'csv') {
    const csv = toCSV(filteredHistory);
    downloadFile(csv, generateFilename('csv'), 'text/csv;charset=utf-8;');
  } else {
    const json = toJSON(filteredHistory);
    downloadFile(json, generateFilename('json'), 'application/json');
  }
  isExportModalOpen.value = false;
};

const filterOptions = [
  { label: 'Semua Kata', value: 'all' },
  { label: '2 Huruf', value: '2' },
  { label: '3 Huruf', value: '3' },
  { label: '4+ Huruf', value: '4+' },
];

const handleSelectFilter = (val: string) => {
  gameStore.getState().setWordLengthFilter(val as any);
  isFilterModalOpen.value = false;
};

// Data chip filter untuk desktop (multi-select)
const desktopChips = computed(() => {
  if (state.value.mode === 'word') {
    return [{ label: 'Semua Kata', value: 'all' as RowType }];
  }
  return [
    { label: 'A', value: 'a' as RowType },
    { label: 'K', value: 'k' as RowType },
    { label: 'S', value: 's' as RowType },
    { label: 'T', value: 't' as RowType },
    { label: 'N', value: 'n' as RowType },
    { label: 'H', value: 'h' as RowType },
    { label: 'M', value: 'm' as RowType },
    { label: 'Y', value: 'y' as RowType },
    { label: 'R', value: 'r' as RowType },
    { label: 'W', value: 'w' as RowType },
    { label: 'ん', value: 'nn' as RowType },
    { label: 'Semua', value: 'all' as RowType },
  ];
});

// Label untuk tombol filter di mobile
const mobileFilterLabel = computed(() => {
  const rows = state.value.selectedRows;
  if (rows.length === 0 || rows.length === 11) return 'Semua';
  return rows.map(r => {
    const found = desktopChips.value.find(c => c.value === r);
    return found ? found.label : r;
  }).join(', ');
});

const personalBest = computed(() => {
  const currentState = state.value;
  if (currentState.history.length === 0) return null;
  
  const relevant = currentState.history.filter(h => {
    return h.mode === currentState.mode && 
           h.timerMode === currentState.timerMode && 
           h.itemCount === (currentState.mode === 'char' ? currentState.maxChars : currentState.maxWords) &&
           (currentState.mode === 'char' || h.wordFilter === currentState.wordLengthFilter);
  });
  
  if (relevant.length === 0) return null;
  return Math.min(...relevant.map(h => h.avgMsPerItem));
});

const fontClass = computed(() => {
  const isWord = state.value.mode === 'word';
  const count = state.value.activeItems.length;
  if (isWord) {
    if (count <= 10) return 'text-3xl md:text-4xl lg:text-5xl';
    if (count <= 25) return 'text-2xl md:text-3xl lg:text-4xl';
    return 'text-xl md:text-2xl lg:text-3xl';
  }
  if (count <= 10) return 'text-5xl md:text-6xl';
  if (count <= 25) return 'text-4xl md:text-5xl';
  return 'text-3xl md:text-4xl';
});

const gridClass = computed(() => {
  if (state.value.mode === 'word') {
    return state.value.useCardStyle
      ? 'grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5'
      : 'grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6';
  }
  return state.value.useCardStyle
    ? 'grid-cols-4 sm:grid-cols-5 md:grid-cols-6 lg:grid-cols-8'
    : 'grid-cols-5 sm:grid-cols-6 md:grid-cols-8 lg:grid-cols-10';
});
</script>

<template>
  <div class="min-h-screen bg-slate-50 p-3 sm:p-4 md:p-8 font-sans">
    
    <header class="text-center mb-4 sm:mb-6">
      <h1 class="text-2xl md:text-3xl font-bold text-slate-800 mb-1">Hiragana Drill</h1>
      <p class="text-slate-500 text-xs sm:text-sm">Latihan membaca cepat</p>
      
      <div 
        v-if="state.isPlaying" 
        class="mt-3 text-2xl sm:text-3xl font-mono font-bold inline-block px-4 py-1 rounded-full transition-colors duration-300"
        :class="[
          state.isPaused 
            ? 'text-green-600 bg-green-50' 
            : (state.timerMode === 'countdown' && state.timeLeft <= 10)
              ? 'text-red-600 bg-red-50 animate-pulse' 
              : 'text-blue-600 bg-blue-50'
        ]"
      >
        <template v-if="state.timerMode === 'countdown'">
          ⏱ {{ state.timeLeft }} dtk
        </template>
        <template v-else>
          ⏱ {{ formatTime(state.elapsedMs) }}
        </template>
        <span v-if="state.isPaused" class="text-xs sm:text-sm font-sans font-normal ml-2 text-green-700">(Paused)</span>
      </div>
    </header>

    <div class="flex justify-center mb-4 sm:mb-6">
      <div class="bg-white rounded-lg shadow-sm p-1 inline-flex border border-gray-100">
        <button 
          @click="gameStore.getState().setMode('char')"
          class="px-4 py-2 rounded-md text-sm font-medium transition-colors"
          :class="state.mode === 'char' ? 'bg-blue-100 text-blue-700' : 'text-gray-600 hover:bg-gray-50'"
          :disabled="state.isPlaying"
        >
          Huruf
        </button>
        <button 
          @click="gameStore.getState().setMode('word')"
          class="px-4 py-2 rounded-md text-sm font-medium transition-colors"
          :class="state.mode === 'word' ? 'bg-blue-100 text-blue-700' : 'text-gray-600 hover:bg-gray-50'"
          :disabled="state.isPlaying"
        >
          Kata
        </button>
      </div>
    </div>

    <AppToolbar 
      :state="state"
      :maxAvailable="maxAvailable"
      :on-start="handleStart"
      :on-stop="handleFinish" 
      :on-toggle-pause="handleTogglePause"
      :on-restart="handleRestartClick" 
      @update:max-items="(val) => gameStore.getState().setMaxItems(val)"
      @update:time-input="(val) => gameStore.getState().setTimeInput(val)"
      @toggle-romaji="() => gameStore.getState().toggleRomaji()"
      @toggle-card-style="() => gameStore.getState().toggleCardStyle()"
      @toggle-shuffle="() => gameStore.getState().toggleShuffle()"
      @open-filter-modal="isFilterModalOpen = true"
      @set-timer-mode="(mode) => gameStore.getState().setTimerMode(mode)"
    />

    <main class="max-w-6xl mx-auto">
      
      <FinishScreen 
        v-if="state.lastRecord" 
        :stats="state.lastRecord"
        @replay="handleReset"
        @viewHistory="isHistoryModalOpen = true"
      />

      <div v-if="!state.isPlaying && !state.lastRecord && state.timeLeft > 0" class="bg-white p-3 sm:p-4 md:p-6 rounded-xl shadow-sm mb-6">
        
        <!-- Header Checklist + Filter -->
        <div class="flex flex-col sm:flex-row sm:justify-between sm:items-center gap-3 mb-4">
          <h2 class="font-semibold text-slate-700 text-sm sm:text-base">
            Pilih {{ state.mode === 'char' ? 'Huruf' : 'Kata' }}
            <span class="ml-1 text-xs bg-blue-100 text-blue-700 px-2 py-1 rounded-md">
              {{ state.selectedItems.length }} / {{ state.mode === 'char' ? state.maxChars : state.maxWords }}
            </span>
          </h2>

          <!-- Tombol Filter (Mobile) -->
          <button 
            @click="isRowFilterModalOpen = true"
            class="sm:hidden flex items-center gap-1 text-sm bg-slate-100 text-slate-600 px-3 py-1.5 rounded-lg hover:bg-slate-200"
          >
            <Filter :size="14" /> 
            {{ mobileFilterLabel }}
          </button>

          <!-- Chip Filter (Desktop/Tablet) - Multi-select -->
          <div class="hidden sm:flex flex-wrap gap-1">
            <button
              v-for="chip in desktopChips"
              :key="chip.value"
              @click="handleToggleRow(chip.value)"
              class="px-2 py-1 rounded-md text-xs font-medium transition-colors border"
              :class="(chip.value === 'all' 
                ? state.selectedRows.length === 11 
                : state.selectedRows.includes(chip.value))
                ? 'bg-blue-100 text-blue-700 border-blue-300' 
                : 'bg-white text-slate-600 border-slate-200 hover:border-slate-300'"
            >
              {{ chip.label }}
            </button>
          </div>
        </div>
        
        <div class="flex flex-wrap gap-1.5 sm:gap-2">
          <button
            v-for="item in currentData"
            :key="item.id"
            @click="handleToggleItem(item)"
            :disabled="isDisabled(item.id)"
            class="rounded-lg border-2 font-bold transition-all duration-200"
            :class="[
              state.mode === 'char' ? 'w-10 h-10 sm:w-12 sm:h-12 text-lg sm:text-xl' : 'px-2 py-1.5 sm:px-3 sm:py-2 text-sm sm:text-base',
              isSelected(item.id) 
                ? 'border-blue-500 bg-blue-50 text-blue-600 shadow-sm' 
                : 'border-slate-200 text-slate-600 hover:border-slate-300 bg-white',
              isDisabled(item.id) ? 'opacity-30 cursor-not-allowed' : ''
            ]"
          >
            {{ item.text }}
          </button>
        </div>
      </div>

      <div 
        v-if="state.isPlaying" 
        class="grid gap-2 md:gap-4 animate-in fade-in slide-in-from-bottom-4 duration-500"
        :class="gridClass"
      >
        <HiraganaItem 
          v-for="item in state.activeItems" 
          :key="item.id" 
          :data="item" 
          :show-romaji="state.showRomaji"
          :use-card-style="state.useCardStyle"
          :class="fontClass"
        />
      </div>

      <div v-if="!state.isPlaying && !state.lastRecord && state.timeLeft === 0 && state.timerMode === 'countdown'" class="text-center mt-12 animate-in fade-in zoom-in duration-300">
        <div class="inline-block bg-white p-6 sm:p-8 rounded-2xl shadow-lg">
          <h2 class="text-2xl sm:text-3xl font-bold text-slate-800 mb-2">Waktu Habis! ⏰</h2>
          <p class="text-slate-500 mb-6 text-sm sm:text-base">Latihan gagal (tidak disimpan).</p>
          <button 
            @click="handleReset" 
            class="bg-blue-600 text-white px-6 sm:px-8 py-2 sm:py-3 rounded-xl font-bold hover:bg-blue-700 transition-colors shadow-md shadow-blue-200"
          >
            Main Lagi
          </button>
        </div>
      </div>

      <div v-if="!state.isPlaying && !state.lastRecord && state.timeLeft > 0" class="mt-8 text-center animate-in fade-in duration-500">
        <button @click="isHistoryModalOpen = true" class="inline-flex items-center gap-2 bg-white px-4 py-2 rounded-xl shadow-sm border border-gray-100 text-sm font-medium text-gray-600 hover:bg-gray-50 transition-colors">
          📜 History ({{ state.history.length }})
          <span v-if="personalBest" class="text-yellow-600 bg-yellow-50 px-2 py-0.5 rounded text-xs font-bold">🏆 Rekor: {{ (personalBest / 1000).toFixed(2) }}s/item</span>
        </button>
      </div>

    </main>

    <AppModal 
      :is-open="isModalOpen" 
      title="Peringatan" 
      message="Pilih minimal satu item sebelum memulai latihan!"
      @close="isModalOpen = false"
    />

    <SelectModal 
      :is-open="isFilterModalOpen"
      title="Filter Panjang Kata"
      :options="filterOptions"
      :selected-value="state.wordLengthFilter"
      @close="isFilterModalOpen = false"
      @select="handleSelectFilter"
    />

    <HistoryModal 
      :is-open="isHistoryModalOpen" 
      :history="state.history" 
      @close="isHistoryModalOpen = false"
      @delete="handleDeleteHistory"
      @clearAll="handleClearAllHistory"
      @open-export="isExportModalOpen = true"
    />

    <ExportModal 
      :is-open="isExportModalOpen"
      :history="state.history"
      @close="isExportModalOpen = false"
      @export="handleExport"
    />

    <ConfirmModal 
      :is-open="isRestartModalOpen"
      title="Mulai Ulang Latihan?"
      message="Timer akan direset dan urutan kartu akan diacak ulang (jika mode acak aktif). Latihan saat ini tidak akan disimpan."
      confirmText="Ya, Mulai Ulang"
      cancelText="Batal"
      confirmColor="orange"
      :icon="RotateCcw"
      @close="isRestartModalOpen = false"
      @confirm="confirmRestart"
    />

    <!-- Modal Filter Baris BARU (Multi-select) -->
    <RowFilterModal 
      :is-open="isRowFilterModalOpen"
      :selected-rows="state.selectedRows"
      :is-word-mode="state.mode === 'word'"
      @close="isRowFilterModalOpen = false"
      @toggle="handleToggleRow"
    />
  </div>
</template>

<script setup lang="ts">
import { Play, Pause, Check, LayoutGrid, AlignJustify, Timer, Filter, RotateCcw } from 'lucide-vue-next';
import type { GameState, TimerMode } from '../types';

defineProps<{
  state: GameState;
  maxAvailable: number;
  onStart: () => void;
  onStop: () => void;
  onTogglePause: () => void;
  onRestart: () => void;
}>();

const emit = defineEmits<{
  (e: 'update:maxItems', val: number): void;
  (e: 'update:timeInput', val: number): void;
  (e: 'toggleRomaji'): void;
  (e: 'toggleCardStyle'): void;
  (e: 'toggleShuffle'): void;
  (e: 'openFilterModal'): void;
  (e: 'setTimerMode', mode: TimerMode): void;
}>();

const handleMaxItemsChange = (e: Event) => {
  const val = Number((e.target as HTMLInputElement).value);
  if (val > 0) emit('update:maxItems', val);
};

const handleTimeInputChange = (e: Event) => {
  emit('update:timeInput', Number((e.target as HTMLInputElement).value));
};
</script>

<template>
  <div class="bg-white p-3 sm:p-4 rounded-xl shadow-sm mb-6 flex flex-wrap gap-3 sm:gap-4 items-center justify-center max-w-5xl mx-auto">
    
    <!-- Input Jumlah -->
    <div class="flex items-center gap-1 border border-gray-300 rounded-lg px-2 bg-gray-50" :class="{ 'opacity-50': state.isPlaying }">
      <span class="text-xs font-semibold text-gray-500">Jml</span>
      <input 
        type="number" 
        :value="state.mode === 'char' ? state.maxChars : state.maxWords" 
        @change="handleMaxItemsChange"
        :disabled="state.isPlaying"
        class="w-12 sm:w-14 p-2 bg-transparent outline-none text-center text-sm font-bold text-gray-700"
        min="1"
        :max="maxAvailable"
      />
    </div>

    <!-- Toggle Timer Mode -->
    <div class="flex items-center border border-gray-300 rounded-lg p-1 bg-gray-50" :class="{ 'opacity-50': state.isPlaying }">
      <button 
        @click="emit('setTimerMode', 'countdown')"
        :disabled="state.isPlaying"
        class="px-2 py-1 rounded text-xs font-medium transition-colors"
        :class="state.timerMode === 'countdown' ? 'bg-white shadow-sm text-blue-600' : 'text-gray-500 hover:text-gray-700'"
      >
        Countdown
      </button>
      <button 
        @click="emit('setTimerMode', 'stopwatch')"
        :disabled="state.isPlaying"
        class="px-2 py-1 rounded text-xs font-medium transition-colors"
        :class="state.timerMode === 'stopwatch' ? 'bg-white shadow-sm text-blue-600' : 'text-gray-500 hover:text-gray-700'"
      >
        Stopwatch
      </button>
    </div>

    <!-- Input Timer (Hanya Countdown) -->
    <div v-if="state.timerMode === 'countdown'" class="flex items-center gap-1 border border-gray-300 rounded-lg px-2 bg-gray-50" :class="{ 'opacity-50': state.isPlaying }">
      <Timer :size="16" class="text-gray-500" />

];

});

    filter: selectedFilter.value,
    filteredHistory: filteredHistory.value
  });

// src/lib/utils.ts
import type { StudyItem, WordLengthFilter } from '../types';

export const shuffleArray = (array: StudyItem[]): StudyItem[] => {
  const newArr = [...array];
  for (let i = newArr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [newArr[i], newArr[j]] = [newArr[j], newArr[i]];
  }
  return newArr;
};

export const isSameOrder = (arr1: StudyItem[], arr2: StudyItem[]): boolean => {
  if (arr1.length !== arr2.length) return false;
  return arr1.every((val, index) => val.id === arr2[index].id);
};

export const getFilteredWords = (filter: WordLengthFilter): StudyItem[] => {
  if (filter === 'all') return WORD_DATA;
  if (filter === '2') return WORD_DATA.filter((w) => w.text.length === 2);
  if (filter === '3') return WORD_DATA.filter((w) => w.text.length === 3);
  return WORD_DATA.filter((w) => w.text.length >= 4);
};

// Fungsi baru untuk format stopwatch: MM:SS.mmm
export const formatTime = (ms: number): string => {
  const totalSec = Math.floor(ms / 1000);
  const m = Math.floor(totalSec / 60);
  const s = totalSec % 60;
  const mm = Math.floor(ms % 1000);
  return `${m > 0 ? m + ':' : ''}${s.toString().padStart(2, '0')}.${mm.toString().padStart(3, '0')}`;
};

    });
  }

// 2. Fungsi Menyimpan History (dengan batas 100)
export const saveHistory = async (entry: Omit<HistoryEntry, 'id'>): Promise<HistoryEntry> => {
  const id = await db.histories.add(entry as HistoryEntry);
  
  // Cek jika lebih dari 100, hapus yang paling lama
  const count = await db.histories.count();
  if (count > 100) {
    const oldest = await db.histories.orderBy('timestamp').first();
    if (oldest && oldest.id) {
      await db.histories.delete(oldest.id);
    }
  }
  
  return { ...entry, id } as HistoryEntry;
};

  return await db.histories.orderBy('timestamp').reverse().toArray();
};

};

// 6. Fungsi Cek Personal Best (PB)
export const checkPersonalBest = async (
  mode: GameMode,
  timerMode: TimerMode,
  itemCount: number,
  wordFilter: WordLengthFilter | undefined,
  currentAvgMs: number
): Promise<boolean> => {
  
  // Ambil semua history dengan kriteria yang sama
  const filtered = await db.histories
    .where({ mode, timerMode, itemCount })
    .toArray();

  // Filter wordFilter (karena Dexie tidak bisa query undefined dengan mudah)
  const relevant = filtered.filter(h => h.wordFilter === wordFilter);

  if (relevant.length === 0) return true; // Jika belum ada history, berarti PB pertama

  // Cari waktu rata-rata terbaik (terkecil)
  const bestAvg = Math.min(...relevant.map(h => h.avgMsPerItem));
  
  return currentAvgMs < bestAvg;
};

// Escape CSV: bungkus dengan quote jika mengandung koma/quote/newline
const escapeCSV = (val: string | number | boolean | undefined): string => {
  if (val === undefined || val === null) return '';
  const str = String(val);
  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
    return `"${str.replace(/"/g, '""')}"`;
  }
  return str;
};

// src/store/gameStore.ts
import { createStore } from 'zustand/vanilla';
import type { GameState, RowType } from '../types';

import { getFilteredWords, shuffleArray } from '../lib/utils';

const getRowSlice = (row: RowType): [number, number] | null => {
  switch (row) {
    case 'a': return [0, 5];
    case 'k': return [5, 10];
    case 's': return [10, 15];
    case 't': return [15, 20];
    case 'n': return [20, 25];
    case 'h': return [25, 30];
    case 'm': return [30, 35];
    case 'y': return [35, 38];
    case 'r': return [38, 43];
    case 'w': return [43, 45];
    case 'nn': return [45, 46];
    default: return null;
  }
};

const getCharsFromRows = (rows: RowType[]): typeof HIRAGANA_DATA => {
  if (rows.length === 0 || rows.includes('all')) return HIRAGANA_DATA;
  const result: typeof HIRAGANA_DATA = [];
  const order: RowType[] = ['a', 'k', 's', 't', 'n', 'h', 'm', 'y', 'r', 'w', 'nn'];
  for (const row of order) {
    if (rows.includes(row)) {
      const slice = getRowSlice(row);
      if (slice) {
        result.push(...HIRAGANA_DATA.slice(slice[0], slice[1]));
      }
    }
  }
  return result;
};

export const gameStore = createStore<GameState>((set, get) => ({
  mode: 'char',
  maxChars: 5,
  maxWords: 5,
  wordLengthFilter: 'all',
  selectedRows: ['a'],
  selectedItems: HIRAGANA_DATA.slice(0, 5),
  activeItems: [],
  showRomaji: false,
  useCardStyle: true,
  isShuffleEnabled: true,
  timeInput: 10,
  timeLeft: 10,
  isPlaying: false,
  isPaused: false,
  
  timerMode: 'countdown',
  startTimestamp: 0,
  elapsedMs: 0,
  pausedAccumMs: 0,
  pauseStartTimestamp: 0,
  history: [],
  lastRecord: null,

  setMode: (mode) => set((state) => {
    const sourceData = mode === 'char' ? getCharsFromRows(state.selectedRows) : getFilteredWords(state.wordLengthFilter);
    const limit = mode === 'char' ? state.maxChars : state.maxWords;
    const newMax = Math.min(limit, sourceData.length);
    return { 
      mode, 
      selectedRows: mode === 'char' ? state.selectedRows : ['all'],
      selectedItems: sourceData.slice(0, newMax), 
      activeItems: [], 
      isPlaying: false, 
      isPaused: false, 
      lastRecord: null, 
      timeLeft: state.timeInput 
    };
  }),

  setMaxItems: (num) => set((state) => {
    const sourceData = state.mode === 'char' ? getCharsFromRows(state.selectedRows) : getFilteredWords(state.wordLengthFilter);
    const validNum = Math.max(1, Math.min(num, sourceData.length));
    if (state.mode === 'char') {
      return { maxChars: validNum, selectedItems: state.selectedItems.slice(0, validNum), lastRecord: null };
    } else {
      return { maxWords: validNum, selectedItems: state.selectedItems.slice(0, validNum), lastRecord: null };
    }
  }),

  setWordLengthFilter: (filter) => set((state) => {
    if (state.mode !== 'word') return { wordLengthFilter: filter };
    const sourceData = getFilteredWords(filter);
    const newMax = Math.min(state.maxWords, sourceData.length);
    return { wordLengthFilter: filter, selectedRows: ['all'], selectedItems: sourceData.slice(0, newMax), lastRecord: null };
  }),

  toggleRowFilter: (row) => set((state) => {
    if (state.mode !== 'char') {
      return { selectedRows: ['all'] };
    }

    let newRows: RowType[] = [];
    const currentRows = state.selectedRows.filter(r => r !== 'all');

    if (row === 'all') {
      newRows = ['a', 'k', 's', 't', 'n', 'h', 'm', 'y', 'r', 'w', 'nn'];
    } else if (currentRows.includes(row)) {
      newRows = currentRows.filter(r => r !== row);
    } else {
      newRows = [...currentRows, row];
    }

    if (newRows.length === 0) {
      newRows = ['all'];
    }

    // PERBAIKAN: maxChars otomatis mengikuti total huruf tersedia
    const availableChars = getCharsFromRows(newRows);
    const newMax = availableChars.length;
    
    return {
      selectedRows: newRows,
      maxChars: newMax,
      selectedItems: availableChars.slice(0, newMax),
      lastRecord: null
    };
  }),

  setTimerMode: (mode) => set((state) => ({ timerMode: mode, timeLeft: state.timeInput, lastRecord: null })),

  toggleItem: (item) => set((state) => {
    const isSelected = state.selectedItems.find((c) => c.id === item.id);
    if (isSelected) {
      return { selectedItems: state.selectedItems.filter((c) => c.id !== item.id) };
    }
    const limit = state.mode === 'char' ? state.maxChars : state.maxWords;
    if (state.selectedItems.length >= limit) return state;
    return { selectedItems: [...state.selectedItems, item] };
  }),

  toggleRomaji: () => set((state) => ({ showRomaji: !state.showRomaji })),
  toggleCardStyle: () => set((state) => ({ useCardStyle: !state.useCardStyle })),
  toggleShuffle: () => set((state) => ({ isShuffleEnabled: !state.isShuffleEnabled })),

  setTimeInput: (time) => set({ timeInput: time, timeLeft: time }),
  setActiveItems: (items) => set({ activeItems: items }),

  loadHistory: async () => {
    const hist = await getHistory();
    set({ history: hist });
  },

  setHistoryData: (history, lastRecord) => set({ history, lastRecord }),

  stopAndSetRecord: (record) => set({ isPlaying: false, isPaused: false, lastRecord: record }),

  startGame: () => {
    const state = get();
    if (state.selectedItems.length === 0) return false;
    set({ 
      isPlaying: true, 
      isPaused: false, 
      timeLeft: state.timeInput, 
      startTimestamp: Date.now(),
      elapsedMs: 0,
      pausedAccumMs: 0,
      pauseStartTimestamp: 0,
      lastRecord: null
    });
    return true;
  },

  restartGame: () => {
    const state = get();
    if (!state.isPlaying) return;
    
    let newActive = state.activeItems;
    if (state.isShuffleEnabled) {
      newActive = shuffleArray(state.selectedItems);
    }

    set({
      activeItems: newActive,
      timeLeft: state.timeInput,
      startTimestamp: Date.now(),
      elapsedMs: 0,
      pausedAccumMs: 0,
      pauseStartTimestamp: 0,
      isPaused: false,
      lastRecord: null
    });
  },

  tickTimer: () => set((state) => {
    if (state.isPaused || !state.isPlaying) return state;
    const now = Date.now();
    const realElapsed = now - state.startTimestamp - state.pausedAccumMs;

    if (state.timerMode === 'countdown') {
      const remaining = Math.max(0, state.timeInput - Math.floor(realElapsed / 1000));
      if (remaining === 0) {
        return { timeLeft: 0, isPlaying: false };
      }
      return { timeLeft: remaining };
    } else {
      return { elapsedMs: realElapsed };
    }
  }),

  togglePause: () => set((state) => {
    if (!state.isPlaying || (state.timerMode === 'countdown' && state.timeLeft === 0)) return state;
    if (state.isPaused) {
      const pauseDuration = Date.now() - state.pauseStartTimestamp;
      return { isPaused: false, pausedAccumMs: state.pausedAccumMs + pauseDuration, pauseStartTimestamp: 0 };
    } else {
      return { isPaused: true, pauseStartTimestamp: Date.now() };
    }
  }),

  resetGame: () => set((state) => ({ 
    isPlaying: false, 
    isPaused: false,
    timeLeft: state.timeInput,
    elapsedMs: 0,
    startTimestamp: 0,
    pausedAccumMs: 0,
    pauseStartTimestamp: 0,
    lastRecord: null,
    activeItems: [] 
  })),
}));
